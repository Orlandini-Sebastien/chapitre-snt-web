            <div class="competence-card">
                <div class="competence-icon">🎨</div>
                <h3>CSS</h3>
                <div class="competence-bar">
                    <div class="competence-progress" data-progress="75"></div>
                </div>
                <span class="competence-percent">75%</span>
            </div>
            
            <div class="competence-card">
                <div class="competence-icon">⚡</div>
                <h3>JavaScript</h3>
                <div class="competence-bar">
                    <div class="competence-progress" data-progress="65"></div>
                </div>
                <span class="competence-percent">65%</span>
            </div>
        </div>
    </section>
    
    <!-- JavaScript -->
    <script>
        // Animation des barres de compétences au chargement
        window.addEventListener('load', function() {
            animerCompetences();
        });
        
        function animerCompetences() {
            let barres = document.querySelectorAll('.competence-progress');
            barres.forEach(barre => {
                let progress = barre.getAttribute('data-progress');
                setTimeout(() => {
                    barre.style.width = progress + '%';
                }, 500);
            });
        }
        
        function scrollToAbout() {
            window.location.href = "about.html";
        }
        
        // Toggle thème sombre/clair
        function toggleTheme() {
            document.body.classList.toggle('dark-theme');
            let bouton = document.getElementById('theme-toggle');
            bouton.innerHTML = document.body.classList.contains('dark-theme') ? '☀️' : '🌙';
            
            // Sauvegarder le thème dans le localStorage
            let isDark = document.body.classList.contains('dark-theme');
            localStorage.setItem('darkTheme', isDark);
        }
        
        // Charger le thème sauvegardé
        window.addEventListener('load', function() {
            if (localStorage.getItem('darkTheme') === 'true') {
                document.body.classList.add('dark-theme');
                document.getElementById('theme-toggle').innerHTML = '☀️';
            }
        });
    </script>
</body>
</html>

3. FORMULAIRE AVEC VALIDATION (contact.html) :

<section class="contact-form">
    <h2>Contactez-moi</h2>
    <form onsubmit="return validerFormulaire()" class="formulaire">
        <div class="form-groupe">
            <label for="nom">Nom *</label>
            <input type="text" id="nom" required>
        </div>
        
        <div class="form-groupe">
            <label for="email">Email *</label>
            <input type="email" id="email" required>
        </div>
        
        <div class="form-groupe">
            <label for="message">Message *</label>
            <textarea id="message" rows="5" required></textarea>
        </div>
        
        <button type="submit" class="btn-envoyer">Envoyer</button>
    </form>
</section>

<script>
function validerFormulaire() {
    let nom = document.getElementById('nom').value.trim();
    let email = document.getElementById('email').value.trim();
    let message = document.getElementById('message').value.trim();
    
    // Validation des champs
    if (nom === '' || email === '' || message === '') {
        alert('Veuillez remplir tous les champs obligatoires');
        return false;
    }
    
    // Validation email simple
    let emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!emailRegex.test(email)) {
        alert('Veuillez entrer un email valide');
        return false;
    }
    
    // Succès
    alert('Merci ' + nom + ' ! Votre message a été envoyé.');
    
    // Reset du formulaire
    document.querySelector('.formulaire').reset();
    
    return false; // Empêche l'envoi réel
}
</script>
