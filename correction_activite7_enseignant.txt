function genererCouleur() {
    // Générer une couleur hexadécimale aléatoire
    let lettres = '0123456789ABCDEF';
    let couleur = '#';
    for (let i = 0; i < 6; i++) {
        couleur += lettres[Math.floor(Math.random() * 16)];
    }
    
    // Appliquer la couleur
    let zone = document.getElementById('zone-couleur');
    zone.style.backgroundColor = couleur;
    zone.style.color = 'white';
    zone.innerHTML = 'Couleur : ' + couleur;
    
    // Afficher le code
    document.getElementById('code-couleur').innerHTML = couleur;
}

2. DÉ ÉLECTRONIQUE AVEC ANIMATION :

let historiqueDesLances = [];

function lancerDe() {
    // Animation du lancer
    let affichage = document.getElementById('affichage-de');
    let animations = ['⚀', '⚁', '⚂', '⚃', '⚄', '⚅'];
    let compteur = 0;
    
    let animation = setInterval(function() {
        affichage.innerHTML = animations[compteur % 6];
        compteur++;
        
        if (compteur > 10) {
            clearInterval(animation);
            // Résultat final
            let resultat = Math.floor(Math.random() * 6) + 1;
            affichage.innerHTML = animations[resultat - 1];
            
            // Ajouter à l'historique
            historiqueDesLances.push(resultat);
            if (historiqueDesLances.length > 5) {
                historiqueDesLances.shift();
            }
            document.getElementById('historique-de').innerHTML = historiqueDesLances.join(' - ');
        }
    }, 100);
}

4. JEU DE RAPIDITÉ AVEC TIMER :

let jeuEnCours = false;
let clicsJeu = 0;
let record = 0;

function commencerJeu() {
    if (jeuEnCours) return;
    
    jeuEnCours = true;
    clicsJeu = 0;
    
    // Interface de jeu
    document.getElementById('zone-jeu').innerHTML = 'CLIQUEZ ICI LE PLUS VITE POSSIBLE !';
    document.getElementById('zone-jeu').style.backgroundColor = '#28A745';
    document.getElementById('zone-jeu').style.cursor = 'pointer';
    document.getElementById('bouton-jeu').innerHTML = 'Jeu en cours...';
    document.getElementById('bouton-jeu').disabled = true;
    
    // Ajouter l'événement de clic
    document.getElementById('zone-jeu').onclick = function() {
        if (jeuEnCours) {
            clicsJeu++;
            this.innerHTML = 'Clics : ' + clicsJeu;
        }
    };
    
    // Timer de 10 secondes
    let tempsRestant = 10;
    let timer = setInterval(function() {
        tempsRestant--;
        document.getElementById('zone-jeu').innerHTML = 'Clics : ' + clicsJeu + ' | Temps : ' + tempsRestant + 's';
        
        if (tempsRestant <= 0) {
            clearInterval(timer);
            finirJeu();
        }
    }, 1000);
}

function finirJeu() {
    jeuEnCours = false;
    
    // Résultats
    if (clicsJeu > record) {
        record = clicsJeu;
        document.getElementById('record').innerHTML = record;
        document.getElementById('zone-jeu').innerHTML = '🎉 NOUVEAU RECORD : ' + clicsJeu + ' clics !';
        document.getElementById('zone-jeu').style.backgroundColor = '#FFD700';
    } else {
        document.getElementById('zone-jeu').innerHTML = 'Terminé ! Score : ' + clicsJeu + ' clics';
        document.getElementById('zone-jeu').style.backgroundColor = '#6C757D';
    }
    
    // Reset interface
    document.getElementById('zone-jeu').style.cursor = 'default';
    document.getElementById('zone-jeu').onclick = null;
    document.getElementById('bouton-jeu').innerHTML = 'Rejouer';
    document.getElementById('bouton-jeu').disabled = false;
}

================================================================================
ERREURS FRÉQUENTES ET SOLUTIONS
================================================================================

ERREUR 1 : Timer ne s'arrête pas
Symptôme : Animation continue indéfiniment
Cause : Oubli de clearInterval() ou variable timer incorrecte
Solution :
- Stocker setInterval dans une variable : let timer = setInterval(...)
- Toujours appeler clearInterval(timer) pour arrêter

ERREUR 2 : Math.random() mal utilisé pour les couleurs
Symptôme : Couleurs invalides ou erreur
Cause : Mauvaise génération hexadécimale
Solution :
- Utiliser Math.floor(Math.random() * 16) pour chaque caractère
- Vérifier la boucle for pour 6 caractères

ERREUR 3 : Variables globales non déclarées
Symptôme : Variables perdues entre les fonctions
Cause : Déclaration dans les fonctions au lieu du global
Solution :
- Déclarer en haut du script : let jeuEnCours = false;
- Pas de let dans les fonctions pour ces variables

ERREUR 4 : Événements onclick multiples
Symptôme : Clics comptés plusieurs fois
Cause : Event listeners non supprimés
Solution :
- Définir onclick = null pour supprimer l'événement
- Vérifier l'état avant d'ajouter des listeners

ERREUR 5 : Array methods incompris
Symptôme : Historique qui ne fonctionne pas
Erreurs fréquentes :
- push() vs unshift() confusion
- shift() pour supprimer le premier élément
Solution :
- push() ajoute à la fin, shift() supprime au début
- join(' - ') pour afficher le tableau
